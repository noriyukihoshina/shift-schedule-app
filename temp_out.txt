    <script>
    (function() {
        const results = [];
        function assert(condition, message) {
            if (condition) results.push("PASS: " + message);
            else results.push("FAIL: " + message);
        }

        // 1. applyOutputConflictHighlights による集計列網掛けテスト
        function applyTestHighlights(conflictCells) {
            document.querySelectorAll('.conflict-highlight').forEach(el => el.classList.remove('conflict-highlight'));
            conflictCells.forEach(item => {
                if (item.statType) {
                    const statCells = document.querySelectorAll(
                        `#outputTableBody td[data-s="${item.staffIndex}"][data-stat="${item.statType}"], ` +
                        `#checkTableBody td[data-s="${item.staffIndex}"][data-stat="${item.statType}"]`
                    );
                    statCells.forEach(cell => cell.classList.add('conflict-highlight'));
                }
            });
        }

        applyTestHighlights([{ staffIndex: 1, statType: 'off', reason: '公休日数が9日です' }]);
        const offCellOutput = document.querySelector('#outputTableBody td[data-s="1"][data-stat="off"]');
        const offCellCheck = document.querySelector('#checkTableBody td[data-s="1"][data-stat="off"]');
        const normalOffCell = document.querySelector('#outputTableBody td[data-s="0"][data-stat="off"]');

        assert(offCellOutput.classList.contains('conflict-highlight'), "修正作業エリアのスタッフ1(公休9)の公休セルに網掛けが付与されること");
        assert(offCellCheck.classList.contains('conflict-highlight'), "最終チェックのスタッフ1(公休9)の公休セルに網掛けが付与されること");
        assert(!normalOffCell.classList.contains('conflict-highlight'), "正常(公休8)のスタッフ0の公休セルには網掛けが付かないこと");

        // 2. モーダル開閉と続行/キャンセルテスト
        let pendingAction = null;
        function showModal(info, onProceed) {
            document.getElementById('confirmEditStaffName').textContent = info.staffName;
            document.getElementById('confirmEditOldSym').textContent = info.oldSym;
            document.getElementById('confirmEditNewSym').textContent = info.newSym;
            pendingAction = onProceed;
            document.getElementById('confirmEditFixedModal').classList.add('open');
        }

        let executed = false;
        showModal({ staffName: "スタッフ 01", oldSym: "休", newSym: "○" }, () => {
            executed = true;
        });

        const modal = document.getElementById('confirmEditFixedModal');
        assert(modal.classList.contains('open'), "赤字修正時にモーダルが open 状態になること");
        assert(document.getElementById('confirmEditOldSym').textContent === "休", "変更前がモーダルに表示されること");
        assert(document.getElementById('confirmEditNewSym').textContent === "○", "変更後がモーダルに表示されること");

        // キャンセルをクリック
        pendingAction = null;
        modal.classList.remove('open');
        assert(!executed, "キャンセル時は修正処理が実行されないこと");

        // 続行テスト
        showModal({ staffName: "スタッフ 01", oldSym: "休", newSym: "○" }, () => {
            executed = true;
        });
        // 続行ボタンクリック
        if (pendingAction) {
            pendingAction();
            pendingAction = null;
        }
        modal.classList.remove('open');
        assert(executed, "続行をクリックした時に修正処理が指示通り実行されること");
        assert(!modal.classList.contains('open'), "続行処理後にモーダルが閉じること");

        document.getElementById('test-output').textContent = results.join('\n');
    })();
    </script>
