// CRC32
const zipCrcTable = new Uint32Array(256);
for (let n = 0; n < 256; n++) {
    let c = n;
    for (let k = 0; k < 8; k++) {
        c = ((c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1));
    }
    zipCrcTable[n] = c;
}
function calcZipCrc32(buf) {
    let crc = 0xFFFFFFFF;
    for (let i = 0; i < buf.length; i++) {
        crc = (crc >>> 8) ^ zipCrcTable[(crc ^ buf[i]) & 0xFF];
    }
    return (crc ^ 0xFFFFFFFF) >>> 0;
}

function createZipUint8Array(files) {

        return {
            name: f.name,
            nameBytes,
            dataBytes,
            crc: calcZipCrc32(dataBytes),
            size: dataBytes.length
        };
    });

    let localHeadersSize = 0;
    fileEntries.forEach(f => {
        f.offset = localHeadersSize;
        localHeadersSize += 30 + f.nameBytes.length + f.size;
    });

    let centralDirSize = 0;
    fileEntries.forEach(f => {
        centralDirSize += 46 + f.nameBytes.length;

function toColLetter(colIdx) {
    let temp = colIdx;
    let letter = '';
    while (temp > 0) {
        let rem = (temp - 1) % 26;
        letter = String.fromCharCode(65 + rem) + letter;
        temp = Math.floor((temp - 1) / 26);
    }
    return letter;
}

async function runAllTests() {
    try {
        // 繧ｹ繧ｿ繝・ヵ縺ｮ蟶梧悍繧偵ユ繧ｹ繝郁ｨｭ螳・

        staffList[3].days[15] = '荳・';

        // 1. XLSX繝舌う繝翫Μ逕滓・
        const zipBuf = generateBlankXlsxBytes();
        let binary = '';
        for (let i = 0; i < zipBuf.length; i++) binary += String.fromCharCode(zipBuf[i]);
        const b64 = btoa(binary);

        // 2. 繧ｽ繝ｫ繝舌・螳溯｡後ユ繧ｹ繝茨ｼ亥ｸ梧悍譫縺檎｢ｺ螳壹＆繧後◆迥ｶ諷九〒隗｣縺代ｋ縺具ｼ・
        const inputGrid = [];
        for (let s = 0; s < NUM_STAFF; s++) {
            inputGrid[s] = [];
            for (let d = 0; d < NUM_DAYS; d++) {
                const sym = staffList[s].days[d] || '';
                inputGrid[s][d] = {
                    symbol: sym,
                    isFixed: (sym !== '')
                };
            }
        }

        const solver = new ShiftScheduler(staffList, inputGrid);
        const solveRes = solver.solve();

        if (!solveRes.success) {
            throw new Error("Solver failed: " + (solveRes.errors || []).join('; '));
        }

        // 蟶梧悍譫縺後☆縺ｹ縺ｦ邯ｭ謖√＆繧後※縺・ｋ縺区､懆ｨｼ

    }
}
