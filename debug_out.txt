<script>
    const NUM_STAFF = 60;
    const NUM_DAYS = 28;
    const staffList = [];
    for (let i = 1; i <= NUM_STAFF; i++) {
        let sec = '';
        if (i <= 12) sec = 'PT A';
        else if (i <= 24) sec = 'PT B';
        else if (i <= 36) sec = 'OT A';
        else if (i <= 48) sec = 'OT B';

        staffList.push({
            id: i,
            name: `スタッフ ${String(i).padStart(2, '0')}`,
            isUnused: (i > 50),
            section: sec,
            can8: (i <= 9),
            noEarly: (i === 13),
            noLate: (i === 13),
            noEve: (i === 13 || i === 15 || i === 16),
            canSched: (i <= 8),
            isRole: (i <= 6),
            isFullTime: (i <= 10),
            allow6Consec: (i === 20 || i === 21),
            prevDays: ['○', '○', '休', '○', '○'],
            days: new Array(NUM_DAYS).fill('')
        });
    }

    const scheduler = new ShiftScheduler(staffList, { standardHolidays: 9.0 });
    scheduler.solveBaseSchedule();

    const ptBIndices = staffList.map((st, idx) => (!st.isUnused && st.section === 'PT B') ? idx : -1).filter(idx => idx >= 0);
    const dailyB = [];
    for (let d = 0; d < 28; d++) {
        let w = 0;
        ptBIndices.forEach(s => {
            if (['○', '出', '早', '遅', 'E', '8時'].includes(scheduler.grid[s][d].symbol)) w++;
        });
        dailyB.push(w);
    }

    document.getElementById('debug').textContent = JSON.stringify({
        ptB_count: ptBIndices.length,
        dailyB: dailyB,
        min: Math.min(...dailyB),
        max: Math.max(...dailyB)
    }, null, 2);
</script>
